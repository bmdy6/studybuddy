# 判分引擎设计文档

> 单一模块 `grading.py`，所有判分路径调同一个函数（约束 #4 单一职责，防双实现漂移）。
> 判分在服务端做，**答案永不下发客户端**。

---

## 1. 核心接口

```python
def grade(
    student_answer: str,
    question: Question,
    rule: GradingRule
) -> GradeResult:
    """
    判分主函数，所有调用方统一入口。

    Returns:
        GradeResult(score, is_correct, detail, needs_manual)
    """
```

```python
@dataclass
class GradeResult:
    score: float          # 实际得分，0 到 question.score
    is_correct: Optional[bool]  # True/False/None(None=待人工判)
    detail: str           # 判分说明（用于日志）
    needs_manual: bool    # 是否需要人工打分
```

## 2. 分流逻辑

```
grade(student_answer, question, rule):
    if rule.rule_type == "manual":
        return GradeResult(0, None, "待人工批改", needs_manual=True)

    if student_answer is None or student_answer.strip() == "":
        return GradeResult(0, False, "未作答", needs_manual=False)

    match by rule.matcher:
        "exact"      -> grade_exact(...)
        "regex"      -> grade_regex(...)
        "levenshtein"-> grade_levenshtein(...)
        "float_tol"  -> grade_float_tol(...)
        "sym_expr"   -> grade_sym_expr(...)  # P2

    if matched:
        return GradeResult(question.score, True, detail, False)
    else:
        return GradeResult(0, False, detail, False)
```

**关键：manual 类型直接返回 is_correct=None，不调匹配器。**

## 3. 匹配器实现

### 3.1 exact（精确匹配）

```python
def grade_exact(
    student_answer: str,
    correct_answer: str,
    params: dict
) -> MatchResult:
    """
    去空格/标点，可选大小写忽略。

    params:
        ignore_case: bool = False
        strip_punct: bool = True
        strip_whitespace: bool = True
    """
    a = student_answer
    b = correct_answer

    if params.get("strip_whitespace", True):
        a, b = a.strip(), b.strip()
        a, b = re.sub(r'\s+', '', a), re.sub(r'\s+', '', b)

    if params.get("strip_punct", True):
        a, b = strip_punctuation(a), strip_punctuation(b)

    if params.get("ignore_case", False):
        a, b = a.lower(), b.lower()

    matched = (a == b)
    return MatchResult(matched, f"exact: '{a}' vs '{b}'")
```

### 3.2 regex（正则匹配）

```python
def grade_regex(
    student_answer: str,
    correct_answer: str,  # 此处 correct_answer 是参考答案，pattern 在 params
    params: dict
) -> MatchResult:
    """
    正则匹配。

    params:
        pattern: str (必填)
        ignore_case: bool = False
    """
    flags = re.IGNORECASE if params.get("ignore_case", False) else 0
    pattern = params["pattern"]

    matched = bool(re.fullmatch(pattern, student_answer, flags))
    return MatchResult(matched, f"regex: {pattern} vs '{student_answer}'")
```

### 3.3 levenshtein（模糊匹配）

```python
def grade_levenshtein(
    student_answer: str,
    correct_answer: str,
    params: dict
) -> MatchResult:
    """
    Levenshtein 距离 ≤ tolerance 视为匹配。
    支持多答案别名 accept_aliases。

    params:
        tolerance: int = DEFAULT_LEVENSHTEIN_TOLERANCE (2)
        accept_aliases: list[str] = []
        ignore_case: bool = True
    """
    tolerance = params.get("tolerance", DEFAULT_LEVENSHTEIN_TOLERANCE)
    aliases = params.get("accept_aliases", [])
    ignore_case = params.get("ignore_case", True)

    candidates = [correct_answer] + aliases
    if ignore_case:
        student_answer = student_answer.lower()
        candidates = [c.lower() for c in candidates]

    for candidate in candidates:
        dist = levenshtein(student_answer, candidate)
        if dist <= tolerance:
            return MatchResult(True, f"levenshtein: dist={dist} ≤ {tolerance}")

    return MatchResult(False, f"levenshtein: min dist > {tolerance}")
```

```python
def levenshtein(s1: str, s2: str) -> int:
    """标准 Levenshtein 距离。"""
    if len(s1) < len(s2):
        return levenshtein(s2, s1)
    if len(s2) == 0:
        return len(s1)

    previous_row = range(len(s2) + 1)
    for i, c1 in enumerate(s1):
        current_row = [i + 1]
        for j, c2 in enumerate(s2):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)
            current_row.append(min(insertions, deletions, substitutions))
        previous_row = current_row

    return previous_row[-1]
```

### 3.4 float_tol（浮点容差）

```python
def grade_float_tol(
    student_answer: str,
    correct_answer: str,
    params: dict
) -> MatchResult:
    """
    数学题浮点容差匹配。

    params:
        float_eps: float = DEFAULT_FLOAT_EPS (0.01)
    """
    eps = params.get("float_eps", DEFAULT_FLOAT_EPS)

    try:
        a = float(student_answer.strip())
        b = float(correct_answer.strip())
    except ValueError:
        return MatchResult(False, f"float_tol: 非数字 '{student_answer}'")

    matched = abs(a - b) <= eps
    return MatchResult(matched, f"float_tol: |{a}-{b}|={abs(a-b)} vs eps={eps}")
```

### 3.5 sym_expr（SymPy 表达式等价，P2）

```python
def grade_sym_expr(
    student_answer: str,
    correct_answer: str,
    params: dict
) -> MatchResult:
    """
    数学符号题：判表达式等价而非字符串相等。
    例："(x+1)^2" 与 "x^2+2x+1" 等价。

    P2 实现，P1 不做。
    """
    from sympy import sympify, simplify, SympifyError

    try:
        a = simplify(sympify(student_answer))
        b = simplify(sympify(correct_answer))
        matched = simplify(a - b) == 0
        return MatchResult(matched, f"sym_expr: equivalent={matched}")
    except SympifyError:
        return MatchResult(False, f"sym_expr: 解析失败 '{student_answer}'")
```

## 4. 部分分机制

RELATE 借鉴：部分分是一等公民。当前 P1 匹配器是二元（对/错），部分分留接口：

```python
@dataclass
class MatchResult:
    matched: bool
    detail: str
    correctness: float = 1.0  # 0-1，匹配则 1.0，未来多选可给部分分
```

- 单选/填空/翻译：correctness 0 或 1
- 多选（未来）：选对部分选项 correctness 0.5
- score = question.score * correctness

## 5. 多选题判分（P1 支持）

```python
def grade_multi_choice(
    student_answer: list,
    correct_answer: list,
    params: dict
) -> MatchResult:
    """
    多选：完全匹配得满分，少选得部分分，多选/错选得 0 分。

    params:
        partial_credit: bool = True  # 少选给部分分
    """
    student_set = set(student_answer)
    correct_set = set(correct_answer)

    if student_set == correct_set:
        return MatchResult(True, "multi: 完全匹配", 1.0)

    if not params.get("partial_credit", True):
        return MatchResult(False, "multi: 不完全匹配", 0.0)

    # 少选：只选了正确答案的子集
    if student_set.issubset(correct_set) and len(student_set) > 0:
        correctness = len(student_set) / len(correct_set)
        return MatchResult(False, f"multi: 少选 {correctness}", correctness)

    # 多选或错选
    return MatchResult(False, "multi: 多选或错选", 0.0)
```

## 6. 判分流程（交卷时）

```
submit_exam(exam_record_id, answers):
    for answer in answers:
        question = get_question(answer.question_id)
        rule = get_grading_rule(question.id)

        result = grade(answer.student_answer, question, rule)

        if result.needs_manual:
            # manual：标记待批改
            save_answer(answer, score=0, is_correct=NULL, is_manual_graded=1)
        else:
            # 自动判分：写分数和对错
            score = question.score * result.correctness
            save_answer(answer, score=score, is_correct=result.is_correct)

            if not result.is_correct:
                save_to_error_book(answer, question)

    update_exam_record_status(exam_record_id, "submitted")
    return feedback_by_mode(exam_record_id, exam.feedback_mode)
```

## 7. 人工打分回填流程

```
grade_manually(answer_id, manual_score, manual_comment):
    answer = get_answer(answer_id)
    question = get_question(answer.question_id)

    # 写人工打分字段
    update_answer(answer_id,
        manual_score=manual_score,
        manual_comment=manual_comment,
        score=manual_score,  # 同步到 score
        is_correct=(manual_score >= question.score * MANUAL_PASS_THRESHOLD),
        is_manual_graded=1,
        graded_at=now()
    )

    # 写错题本
    if not is_correct:
        save_to_error_book(answer, question)

    # 检查是否全部判完
    if all_answers_graded(answer.exam_record_id):
        update_exam_record_status(exam_record_id, "graded")
        calculate_total_score(exam_record_id)
```

**MANUAL_PASS_THRESHOLD** 常量：
```python
MANUAL_PASS_THRESHOLD = 0.6  # 人工打分 >= 满分 60% 判对
```

## 8. 反馈门控（feedback_mode）

交卷后按 feedback_mode 决定返回给学生的内容：

| feedback_mode | 客观题对错 | 主观题状态 | 分数 |
|---|---|---|---|
| immediate（练习卷） | ✅ 即时 | "待批改" | 客观题分 |
| after_submit（考试卷） | ✅ 交卷后 | "待批改" | 客观题分，主观题判完再更新 |
| hidden（纯考试） | ❌ 不显示 | ❌ 不显示 | ❌ 不显示 |

**安全：** hidden 模式下 GET /api/exam-record/{id}/result 也按 mode 过滤，不泄答案和对错。

## 9. 测试设计

判分引擎是纯函数，无 DB 依赖，可独立单测（约束 #7 测试可写性）。

```python
# tests/grading/test_exact.py
def test_exact_match_ignore_case():
    result = grade_exact("WENT", "went", {"ignore_case": True, "strip_punct": True})
    assert result.matched is True

def test_exact_mismatch():
    result = grade_exact("goed", "went", {"ignore_case": False, "strip_punct": True})
    assert result.matched is False

# tests/grading/test_levenshtein.py
def test_levenshtein_within_tolerance():
    result = grade_levenshtein("wentt", "went", {"tolerance": 2})
    assert result.matched is True

def test_levenshtein_accept_alias():
    result = grade_levenshtein("goed", "went",
        {"tolerance": 1, "accept_aliases": ["goed"]})
    assert result.matched is True

# tests/grading/test_manual.py
def test_manual_needs_manual_flag():
    result = grade("any answer", question, manual_rule)
    assert result.needs_manual is True
    assert result.is_correct is None
```
