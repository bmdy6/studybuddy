from datetime import datetime
from typing import Any, Optional
from sqlalchemy import JSON, BigInteger, DateTime, Enum, ForeignKey, Integer, String, Text, UniqueConstraint, func
from sqlalchemy.orm import Mapped, mapped_column
from app.database import Base


class KnowledgePoint(Base):
    __tablename__ = "knowledge_points"
    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)
    space_id: Mapped[int] = mapped_column(BigInteger, ForeignKey("spaces.id", ondelete="CASCADE"), nullable=False)
    name: Mapped[str] = mapped_column(String(128), nullable=False)
    parent_id: Mapped[Optional[int]] = mapped_column(BigInteger, ForeignKey("knowledge_points.id"), nullable=True)
    aliases: Mapped[Optional[list[Any]]] = mapped_column(JSON, nullable=True)

    source_event_id: Mapped[Optional[int]] = mapped_column(BigInteger, nullable=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, server_default=func.now())
    __table_args__ = (UniqueConstraint("space_id", "name", name="uk_kp_space_name"),)


class KnowledgePointProgress(Base):
    __tablename__ = "knowledge_point_progress"
    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)
    knowledge_point_id: Mapped[int] = mapped_column(BigInteger, ForeignKey("knowledge_points.id", ondelete="CASCADE"), nullable=False)

    first_learned_at: Mapped[Optional[datetime]] = mapped_column(DateTime, nullable=True)
    updated_at: Mapped[datetime] = mapped_column(DateTime, server_default=func.now(), onupdate=func.now())
    __table_args__ = (UniqueConstraint("knowledge_point_id", "user_id", name="uk_kpp_point_user"),)


class KnowledgeEvent(Base):
    __tablename__ = "knowledge_events"
    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)

# app/features/knowledge/progress.py
"""证据 → confidence 更新。规则与 docs/plans/2026-08-10-知识点掌握度系统-design.md 一致。"""

# 练习加减分（按难度）
STRENGTHEN_POINTS = {"easy": 6, "medium": 8, "hard": 10}
WEAKEN_POINTS = {"easy": 4, "medium": 6, "hard": 8}
STREAK_BONUS = 2  # 连续答对 ≥3 次额外奖励
EXAM_WEIGHT = 0.6
PRACTICE_WEIGHT = 0.4
DECAY_MONTHLY = 5  # 30 天无作答每月衰减


def apply_event(confidence: int, event_type: str, verdict: str,
                difficulty: str = "medium", streak: int = 0) -> int:
    """单条事件更新 confidence（0-100）。"""
    delta = 0
    if verdict == "strengthen":
        delta = STRENGTHEN_POINTS.get(difficulty, 8)
        if streak >= 3:
            delta += STREAK_BONUS
    elif verdict == "weaken":
        delta = -WEAKEN_POINTS.get(difficulty, 6)
    return max(0, min(100, confidence + delta))


def score_from_evidence(evidence_type: str, score_rate: float | None = None,
                        practice_score: int = 0) -> int:
    """综合得分：考试证据（得分率）权重 0.6，练习证据权重 0.4。
    无考试时纯练习分。score_rate 0-1。"""
    if evidence_type == "exam" and score_rate is not None:
        return max(0, min(100, int(round(EXAM_WEIGHT * score_rate * 100 + PRACTICE_WEIGHT * practice_score))))
    return max(0, min(100, practice_score))


def apply_decay(confidence: int, days_since_last: int) -> int:
    """遗忘衰减：30 天无证据后每月 -5。"""
    if days_since_last <= 30:
        return confidence
    months = (days_since_last - 30) // 30
    return max(0, confidence - DECAY_MONTHLY * months)
```

# app/features/knowledge/evidence.py
"""各类学习行为 → knowledge_events。强证据直接更新 progress.confidence。"""
from sqlalchemy.orm import Session
from app.models.knowledge import KnowledgeEvent, KnowledgePointProgress
from app.features.knowledge.progress import apply_event


def record_event(db: Session, user_id: int, knowledge_point_id: int,
                 event_type: str, source_id: int | None, verdict: str,
                 content: str | None = None, difficulty: str = "medium",
                 streak: int = 0) -> KnowledgeEvent:
    """写事件 + 更新 progress.confidence。"""
    event = KnowledgeEvent(
        knowledge_point_id=knowledge_point_id, user_id=user_id,
        event_type=event_type, source_id=source_id, content=content, verdict=verdict,
    )
    db.add(event)

    progress = db.query(KnowledgePointProgress).filter_by(
        knowledge_point_id=knowledge_point_id, user_id=user_id).first()
    if progress is None:
        progress = KnowledgePointProgress(knowledge_point_id=knowledge_point_id, user_id=user_id)
        db.add(progress)
        db.flush()

    progress.confidence = apply_event(progress.confidence, event_type, verdict, difficulty, streak)
    progress.status = _derive_status(progress.confidence)
    progress.last_reviewed_at = func.now()
    db.commit()
    return event


def _derive_status(confidence: int) -> str:
    if confidence >= 85:
        return "graduated"
    if confidence >= 60:
        return "learning"
    if confidence > 0:
        return "learning"
    return "new"
```
